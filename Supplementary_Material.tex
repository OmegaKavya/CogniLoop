\section{Formal Mathematical Derivations}

\subsection{Bayesian Posterior Derivation for DC-BKT Updates}
Let $L_n \in \{0, 1\}$ be the latent binary knowledge state for a given concept at assessment step $n$, where $L_n = 1$ denotes concept mastery and $L_n = 0$ denotes non-mastery. The prior mastery probability is denoted by $P(L_n) = P(L_n = 1)$.

Let $Obs_n \in \{0, 1\}$ be the observable binary response on an item of difficulty tier $a \in \{\text{easy}, \text{medium}, \text{hard}\}$. The observation conditional probabilities parameterized by difficulty-conditioned guess probability $P(G_a)$ and slip probability $P(S_a)$ are defined as:
\begin{align}
P(Obs_n = 1 \mid L_n = 1) &= 1 - P(S_a) \\
P(Obs_n = 1 \mid L_n = 0) &= P(G_a) \\
P(Obs_n = 0 \mid L_n = 1) &= P(S_a) \\
P(Obs_n = 0 \mid L_n = 0) &= 1 - P(G_a)
\end{align}

Applying Bayes' Theorem for a correct observation ($Obs_n = 1$):
\begin{equation}
P(L_n = 1 \mid Obs_n = 1) = \frac{P(Obs_n = 1 \mid L_n = 1) P(L_n = 1)}{P(Obs_n = 1)}
\end{equation}
Expanding the denominator via the law of total probability:
\begin{align}
P(Obs_n = 1) &= P(Obs_n = 1 \mid L_n = 1)P(L_n = 1) \nonumber \\
&\quad + P(Obs_n = 1 \mid L_n = 0)P(L_n = 0) \nonumber \\
&= (1 - P(S_a))P(L_n) + P(G_a)(1 - P(L_n))
\end{align}
Substituting the denominator yields the posterior estimation:
\begin{equation}
P(L_n \mid Obs_n = 1) = \frac{P(L_n)(1 - P(S_a))}{P(L_n)(1 - P(S_a)) + (1 - P(L_n))P(G_a)}
\end{equation}

Similarly, for an incorrect response ($Obs_n = 0$):
\begin{equation}
P(L_n \mid Obs_n = 0) = \frac{P(L_n)P(S_a)}{P(L_n)P(S_a) + (1 - P(L_n))(1 - P(G_a))}
\end{equation}

Accounting for knowledge acquisition during active retrieval practice, the transition Markov model projects mastery to step $n+1$:
\begin{align}
P(L_{n+1}) &= P(L_n \mid Obs_n) + (1 - P(L_n \mid Obs_n)) P(T)
\end{align}
where $P(T) = 0.20$ represents the baseline probability of transitioning from the unlearned to learned state after instruction.

\subsection{Thompson Sampling Posterior Beta Updates with Fractional Observations}
Let $\mathcal{S}$ denote the state space $s = \langle C, M_{\text{bin}} \rangle$ and $\mathcal{A} = \{\text{easy}, \text{medium}, \text{hard}\}$ denote the action space. For each $(s, a)$, the bandit maintains a prior distribution over the expected performance $\mu_{s,a} \in [0, 1]$:
\begin{equation}
f(\mu_{s,a}; \alpha_{s,a}, \beta_{s,a}) = \frac{\Gamma(\alpha_{s,a} + \beta_{s,a})}{\Gamma(\alpha_{s,a})\Gamma(\beta_{s,a})} \mu_{s,a}^{\alpha_{s,a} - 1} (1 - \mu_{s,a})^{\beta_{s,a} - 1}
\end{equation}
Given normalized observation $r = \frac{\text{Score}}{100.0} \in [0, 1]$, the continuous fractional update corresponds to:
\begin{equation}
\alpha_{s,a} \leftarrow \alpha_{s,a} + r, \quad \beta_{s,a} \leftarrow \beta_{s,a} + (1 - r)
\end{equation}
Action selection incorporates difficulty utility weighting $w_a \in \{0.8, 1.0, 1.2\}$:
\begin{equation}
a^* = \arg\max_{a \in \mathcal{A}} \left( \theta_a \times w_a \right), \quad \theta_a \sim \text{Beta}(\alpha_{s,a}, \beta_{s,a})
\end{equation}
